#Medir distancia con TOF 400F usando UART en Raspberry Pi Pico 2
from machine import Pin, UART
import time

# Inicializamos el puerto UART0
uart = UART(0, baudrate=115200, tx=Pin(0), rx=Pin(1))

print("--- Extractor Modbus para TOF400F Activo ---")

while True:
    if uart.any():
        datos = uart.read()
        
        if datos is not None:
            # Recorremos el buffer buscando la cabecera estándar Modbus del sensor: 0x01, 0x03, 0x02
            # Necesitamos al menos 5 bytes para leer la cabecera + los 2 bytes de distancia
            i = 0
            while i <= len(datos) - 5:
                if datos[i] == 1 and datos[i+1] == 3 and datos[i+2] == 2:
                    # Extraemos el byte alto y el byte bajo de la distancia
                    byte_alto = datos[i+3]
                    byte_bajo = datos[i+4]
                    
                    # Calculamos la distancia matemática en milímetros
                    distancia_mm = (byte_alto << 8) | byte_bajo
                    
                    # Filtrado de lecturas: 65535 significa fuera de rango o error
                    if distancia_mm == 65535:
                        print("Distancia: Fuera de rango o error (Max)")
                    else:
                        print(f"Distancia Real: {distancia_mm} mm ({distancia_mm / 10:.1f} cm)")
                    
                    # Saltamos el paquete procesado para buscar el siguiente
                    i += 5
                else:
                    i += 1
                    
    time.sleep_ms(30) # Pausa óptima para la ráfaga de datos del sensor
