"""
TEST STANDALONE - Calibracion del sensor de color TCS34725
Imprime r, g, b, claro en bucle continuo. Pasad el sensor sobre cada
superficie (blanco del tapete, linea naranja, linea azul, y alguno de los
baches/zonas grises que dan falsos positivos) y anotad los rangos que veis
para cada una.
"""
from machine import Pin, I2C
from tcs34725 import TCS34725
import time

i2c0 = I2C(0, sda=Pin(4), scl=Pin(5))
sensor_color = TCS34725(i2c0)
print("Sensor listo. Pasando a lectura continua...")

while True:
    r, g, b, claro = sensor_color.read_rgbc()
    total = r + g + b
    if total > 0:
        pr = r / total
        pg = g / total
        pb = b / total
        print(f"r={r:4d} g={g:4d} b={b:4d} claro={claro:4d}  |  r%={pr:.2f} g%={pg:.2f} b%={pb:.2f}")
    else:
        print(f"r={r:4d} g={g:4d} b={b:4d} claro={claro:4d}  |  (total=0)")
    time.sleep_ms(200)
