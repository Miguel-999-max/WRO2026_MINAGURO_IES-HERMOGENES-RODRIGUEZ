from machine import Pin, UART, PWM
import time

# ==========================================
# 1. CONFIGURACIÓN DE PINES (DOCUMENTO INICIAL)
# ==========================================
pulsador = Pin(2, Pin.IN, Pin.PULL_DOWN)
led_verde = Pin(1, Pin.OUT)
led_rojo = Pin(22, Pin.OUT)

# Comunicación UART0 para el LiDAR TFmini-S
uart0 = UART(0, baudrate=115200, tx=Pin(0), rx=Pin(1))

# Actuadores (PWM a 50Hz)
servo_traccion = PWM(Pin(14))
servo_direccion = PWM(Pin(20))
servo_radar = PWM(Pin(21))

servo_traccion.freq(50)
servo_direccion.freq(50)
servo_radar.freq(50)

# ==========================================
# 2. CONSTANTES DE CALIBRACIÓN REALES
# ==========================================
# Tracción
TRACCION_STOP = 5000
TRACCION_AVANCE_TOPE = 3600
TRACCION_AVANCE_SUAVE = 4200   # Valor intermedio para pruebas iniciales seguras

# Dirección
DIRECCION_CENTRO = 5000
LIMITE_DIR_IZQ = 6190          # Giro máximo izquierdo (38º)
LIMITE_DIR_DER = 3900          # Giro máximo derecho (38º)

# Radar (Servo Lidar)
RADAR_FIJO_DERECHA = 1350      # Clavado mirando a la derecha

# Parámetros del algoritmo de seguimiento
DISTANCIA_OBJETIVO = 15        # SetPoint: Distancia deseada a la pared (cm)
GANANCIA_KP = 85               # Ajustar si la corrección es muy lenta o muy brusca

# ==========================================
# 3. FUNCIONES AUXILIARES
# ==========================================
def leer_tfmini():
    """Vacía el búfer de la UART0 buscando una lectura válida del LiDAR."""
    if uart0.any():
        datos = uart0.read()
        if datos is not None:
            i = len(datos) - 9
            # Recorremos de atrás hacia adelante para tener el dato más fresco
            while i >= 0:
                if datos[i] == 89 and datos[i+1] == 89: # 0x59 0x59
                    fuerza = datos[i+4] + (datos[i+5] << 8)
                    if fuerza >= 100:
                        distancia = datos[i+2] + (datos[i+3] << 8)
                        return distancia
                    break
                i -= 1
    return None

# ==========================================
# 4. FASE DE INICIALIZACIÓN (ESPERA DE BOTÓN)
# ==========================================
print("--- Robot Calibrado e Inicializado ---")
# Asegurar estado estático antes de arrancar
servo_traccion.duty_u16(TRACCION_STOP) 
servo_direccion.duty_u16(DIRECCION_CENTRO)
servo_radar.duty_u16(RADAR_FIJO_DERECHA)

# Bucle de espera (Parpadeo de LEDs)
while pulsador.value() == 0:
    led_rojo.value(1)
    led_verde.value(0)
    time.sleep_ms(100)
    led_rojo.value(0)
    led_verde.value(1)
    time.sleep_ms(100)

led_verde.value(1)
led_rojo.value(0)
print("--- ¡EN MARCHA! ---")

# ==========================================
# 5. BUCLE PRINCIPAL
# ==========================================
distancia_actual = DISTANCIA_OBJETIVO

while True:
    # 1. Ordenar avance al motor (Usamos velocidad suave para testear)
    servo_traccion.duty_u16(TRACCION_AVANCE_TOPE)
    
    # 2. Capturar distancia por hardware
    nueva_distancia = leer_tfmini()
    if nueva_distancia is not None:
        distancia_actual = nueva_distancia

    # 3. Control Proporcional (P)
    error = distancia_actual - DISTANCIA_OBJETIVO
    
    # EXPLICACIÓN DE LA MATEMÁTICA:
    # Si Distancia > 15 -> Error es POSITIVO (El robot está lejos de la pared).
    # Necesitamos girar a la DERECHA. Según su calibración, derecha es BAJAR el PWM.
    # Por lo tanto: CENTRO (5000) - (Error Positivo * KP) = Un valor MENOR a 5000 (Gira a la derecha). ¡Perfecto!
    pulso_dir = DIRECCION_CENTRO - (error * GANANCIA_KP)
    
    # 4. Seguridad: Limitar el rango con sus topes mecánicos reales (3900 a 6190)
    # Como el límite numérico mínimo es el giro derecho (3900) y el máximo es izquierdo (6190):
    pulso_dir = max(LIMITE_DIR_DER, min(pulso_dir, LIMITE_DIR_IZQ))
    
    # 5. Actuar sobre las ruedas direccionales
    servo_direccion.duty_u16(pulso_dir)
    
    # Telemetría por consola para ajuste fino de la GANANCIA_KP
    print(f"Dist={distancia_actual}cm | Error={error} | PulsoDIR={pulso_dir}")
    
    time.sleep_ms(20)
