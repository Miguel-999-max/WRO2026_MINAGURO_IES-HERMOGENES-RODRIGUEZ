from machine import Pin, UART, PWM, I2C
import time
from tcs34725 import TCS34725 

# ==========================================
# 1. CONFIGURACION DE PINES 
# ==========================================
pinBoton = Pin(2, Pin.IN, Pin.PULL_DOWN)
ledV = Pin(1, Pin.OUT)
ledR = Pin(22, Pin.OUT)

# Comunicacion UART0 para el LiDAR TFmini-S
uart0 = UART(0, baudrate=115200, tx=Pin(0), rx=Pin(1))

# Actuadores (PWM a 50Hz)
servoTraccion = PWM(Pin(14))
servoDireccion = PWM(Pin(20))
SERVOcam = PWM(Pin(21))

servoTraccion.freq(50)
servoDireccion.freq(50)
SERVOcam.freq(50)

# ==========================================
# 2. CONSTANTES DE CALIBRACION REALES
# ==========================================
# Servo Direccion
Recto = 5000
Dcha = 3900
Izda = 6190

# Servo Camara / Lidar (Pin 21)
Cam_Centro = 4675
RADAR_FIJO_DERECHA = 1350

# Servo Traccion (Rotacion continua)
Para = 5000
Avanza_tope = 3600
Retrocede_tope = 6250

# Parametros del algoritmo de seguimiento
DISTANCIA_OBJETIVO = 15        # SetPoint a la pared (cm)
GANANCIA_KP = 85               # Sensibilidad de direccion

# Umbrales del sensor de color trasero
UMBRAL_BLANCO_A_COLOR = 500 
PUNTO_CORTE_COLORES = 210   

# Variables de carrera
TOTAL_LINEAS_META = 36         # 12 lineas * 3 vueltas
contadorLineas = 0             
sobre_linea = False            # bandera para control de flanco

# ==========================================
# 3. FUNCIONES AUXILIARES
# ==========================================
def leer_tfmini():
    """Vacia el bufer de la UART0 buscando una lectura valida del LiDAR."""
    if uart0.any():
        datos = uart0.read()
        if datos is not None:
            i = len(datos) - 9
            while i >= 0:
                if datos[i] == 89 and datos[i+1] == 89: # 0x59 0x59
                    fuerza = datos[i+4] + (datos[i+5] << 8)
                    if fuerza >= 100:
                        distancia = datos[i+2] + (datos[i+3] << 8)
                        return distancia
                    break
                i -= 1
    return None

# ==========================================
# 4. FUNCIONES DE LAS MISIONES (ZONA ABIERTA)
# ==========================================
def ejecutar_mision_1_cw():
    """MISION 1: Giro a derechas siguiendo pared con control de lineas por flanco."""
    global contadorLineas, sobre_linea
    print("--- EJECUTANDO MISION 1 CW (GIRO A DERECHAS) ---")

    # Clavamos el radar a la derecha para ver la pared en la carrera
    SERVOcam.duty_u16(RADAR_FIJO_DERECHA)
    
    # --- MOVIMIENTOS INICIALES POR TIEMPO ---
    print("Movimiento Inicial 1: Avanzando a la derecha...")
    servoDireccion.duty_u16(Dcha)
    servoTraccion.duty_u16(Avanza_tope)
    time.sleep_ms(800) 

    print("Movimiento Inicial 2: Avanzando a la izquierda...")
    servoDireccion.duty_u16(Izda)
    servoTraccion.duty_u16(Avanza_tope)
    time.sleep_ms(800) 

    print("Salida completada. Entrando en bucle de carrera...")
    distancia_actual = DISTANCIA_OBJETIVO

    while True:
        # 1. Traccion a tope hacia adelante
        servoTraccion.duty_u16(Avanza_tope)
        
        # 2. Medir la pared derecha con el LiDAR
        nueva_distancia = leer_tfmini()
        if nueva_distancia is not None:
            distancia_actual = nueva_distancia

        # 3. Control Proporcional (P)
        error = distancia_actual - DISTANCIA_OBJETIVO
        pulso_dir = Recto - (error * GANANCIA_KP)
        
        # 4. Seguridad de direccion
        pulso_dir = max(Dcha, min(pulso_dir, Izda))
        servoDireccion.duty_u16(pulso_dir)
        
        # 5. Conteo de lineas por flanco con su libreria
        r, g, b, claro = sensor_color.read_rgbc()
        
        if not sobre_linea:
            if claro < UMBRAL_BLANCO_A_COLOR: # Entramos a la linea (Flanco bajada)
                contadorLineas += 1
                sobre_linea = True
                ledR.value(1) # testigo visual
                print(f"¡LINEA DETECTADA! -> Total lineas: {contadorLineas}/36")
        else:
            if claro >= UMBRAL_BLANCO_A_COLOR: # Salimos al blanco (Flanco subida)
                sobre_linea = False
                ledR.value(0)

        # 6. Fin de carrera (3 vueltas)
        if contadorLineas >= TOTAL_LINEAS_META:
            print("¡36 lineas completadas! Esperando un segundo para clavar la meta...")
            time.sleep_ms(1000) # delay ajustable final
            
            # Parada total definitiva
            servoTraccion.duty_u16(Para)
            servoDireccion.duty_u16(Recto)
            print("--- CARRERA COMPLETADA CON EXITO ---")
            while True:
                ledR.value(1)
                ledV.value(1)
                time.sleep_ms(500)
                
        time.sleep_ms(20)

def ejecutar_mision_2_ccw():
    """MISION 2: Giro a izquierdas (Se programara en el siguiente paso)."""
    print("--- MISION 2 CCW SELECCIONADA (Falta programar este bloque) ---")
    servoTraccion.duty_u16(Para)

# ==========================================
# 5. FASE DE INICIALIZACION (ESPERA DE BOTON)
# ==========================================
print("--- Robot Unificado Inicializado ---")
servoTraccion.duty_u16(Para) 
servoDireccion.duty_u16(Recto)
SERVOcam.duty_u16(Cam_Centro)

# Inicializar bus I2C0 para el sensor de color
i2c0 = I2C(0, sda=Pin(4), scl=Pin(5))
try:
    sensor_color = TCS34725(i2c0)
    print("¡Sensor TCS34725 listo!")
except Exception as e:
    print("Error sensor color:", e)
    sensor_color = None

# Espera de boton parpadeando led
while pinBoton.value() == 0:
    ledR.value(1)
    ledV.value(0)
    time.sleep_ms(100)
    ledR.value(0)
    ledV.value(1)
    time.sleep_ms(100)

ledV.value(0)
ledR.value(0)
print("--- ¡ARRANQUE INTERNO! ---")

# ==========================================
# 6. SCAN 1 Y DECISION INICIAL
# ==========================================
distFR = 110
while True:
    lectura = leer_tfmini()
    if lectura is not None:
        distFR = lectura
        break
    time.sleep_ms(10)

print(f"Distancia Frontal Inicial: {distFR} cm")

if distFR > 40:
    # Zona Abierta: Vamos marcha atras a tope para pisar la linea
    ledV.value(1) 
    time.sleep_ms(2000)
    ledV.value(0)
    
    servoDireccion.duty_u16(Recto)
    servoTraccion.duty_u16(Retrocede_tope)
    
    linea_detectada = False
    luz_ambiente_final = 700
    
    while not linea_detectada and sensor_color is not None:
        r, g, b, claro = sensor_color.read_rgbc()
        if claro < UMBRAL_BLANCO_A_COLOR:
            servoTraccion.duty_u16(Para) # Frenazo por inercia
            luz_ambiente_final = claro
            linea_detectada = True
        time.sleep_ms(5)
        
    print(f"¡Linea pisada! Valor estatico: {luz_ambiente_final}")
    
    # Saltamos directamente a la funcion de la mision elegida
    if luz_ambiente_final < PUNTO_CORTE_COLORES:
        print("COLOR AZUL -> Iniciando Misión 1")
        ejecutar_mision_1_cw()
    else:
        print("COLOR NARANJA -> Iniciando Misión 2")
        ejecutar_mision_2_ccw()
else:
    # Zona de Obstaculos (Se completara mas adelante)
    ledR.value(1)
    time.sleep_ms(2000)
    ledR.value(0)
    print("Zona Obstaculos Detectada.")
