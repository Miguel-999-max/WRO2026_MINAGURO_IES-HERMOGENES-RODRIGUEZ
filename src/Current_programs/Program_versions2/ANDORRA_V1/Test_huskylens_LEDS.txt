"""
TEST STANDALONE - Huskylens + LEDs
Enciende el LED VERDE si ve un bloque ID2 (verde) y el LED ROJO si ve un
bloque ID1 (rojo). Si ve los dos a la vez, enciende los dos.
Requiere subir tambien huskylens.py a la Pico.

Conexion:
  SDA1 -> GPIO14
  SCL1 -> GPIO15
"""

from machine import Pin, I2C
from huskylens import HuskyLens
import time

ledV = Pin(10, Pin.OUT)
ledR = Pin(22, Pin.OUT)

i2c1 = I2C(1, sda=Pin(14), scl=Pin(15), freq=100000, timeout=100000)
hl = HuskyLens(i2c1)

print("Dispositivos I2C1 encontrados:", [hex(a) for a in i2c1.scan()])
print("Knock:", hl.knock())

while True:
    bloques = hl.get_blocks()

    hay_rojo = any(b.id == 1 for b in bloques)
    hay_verde = any(b.id == 2 for b in bloques)

    ledR.value(1 if hay_rojo else 0)
    ledV.value(1 if hay_verde else 0)

    time.sleep_ms(100)
